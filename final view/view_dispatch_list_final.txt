 create view view_dispatch_list_final as
SELECT
    `aed_elcot`.`po_form`.`po_num` AS `po_num`,
    `aed_elcot`.`po_form`.`po_date` AS `po_date`,
    `aed_elcot`.`po_form`.`unique_id` AS `unique_id`,
    `aed_elcot`.`po_form`.`is_delete` AS `is_delete`,
    `con`.`unique_id` AS `con_unique_id`,
    `con`.`con_contact_name` AS `con_contact_name`,
    `con`.`con_address` AS `con_address`,
    `aed_elcot`.`invoice_creation`.`invoice_auto_id` AS `invoice_auto_id`,
    `aed_elcot`.`invoice_creation`.`invoice_no` AS `invoice_no`,
    `aed_elcot`.`invoice_creation`.`invoice_date` AS `invoice_date`,
    `aed_elcot`.`invoice_creation`.`invoice_qty` AS `invoice_qty`,
    `aed_elcot`.`invoice_creation`.`unit_price` AS `unit_price`,
    `aed_elcot`.`invoice_creation`.`product_unique_id` AS `product_unique_id`,
    `aed_elcot`.`invoice_creation`.`stock_id` AS `stock_id`
FROM
    (
        (
            `aed_elcot`.`po_form`
        JOIN `aed_elcot`.`consignee_details_sub` `con`
        ON
            (
                `aed_elcot`.`po_form`.`unique_id` = `con`.`form_main_unique_id`
            )
        )
    JOIN `aed_elcot`.`invoice_creation` ON
        (
            `aed_elcot`.`po_form`.`unique_id` = `aed_elcot`.`invoice_creation`.`po_unique_id`
        )
    )
WHERE
    `aed_elcot`.`po_form`.`unique_id` <> '' AND `aed_elcot`.`po_form`.`is_delete` = '0'
GROUP BY
    `aed_elcot`.`invoice_creation`.`invoice_auto_id`
