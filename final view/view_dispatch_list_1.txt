Create view view_dispatch_list_1 
AS 
SELECT
    `aed_elcot`.`po_form`.`po_num` AS `po_num`,
    `aed_elcot`.`po_form`.`po_date` AS `po_date`,
    `con`.`con_contact_name` AS `con_contact_name`,
    `con`.`con_address` AS `con_address`
FROM
    (
        `aed_elcot`.`po_form`
    JOIN `aed_elcot`.`consignee_details_sub` `con`
    ON
        (
            `aed_elcot`.`po_form`.`unique_id` = `con`.`form_main_unique_id`
        )
    )
WHERE
    `aed_elcot`.`po_form`.`unique_id` <> '' AND `aed_elcot`.`po_form`.`is_delete` = '0'
