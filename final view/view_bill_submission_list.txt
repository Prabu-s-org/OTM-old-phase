create view view_bill_submission_list as
SELECT
    `inv`.`form_main_unique_id` AS `form_main_unique_id`,
    `inv`.`po_num` AS `po_num`,
    `inv`.`po_date` AS `po_date`,
    `sign_doc`.`installation_reference_no` AS `installation_reference_no`,
    `sign_doc`.`without_bg` AS `proced_without_bg`,
    `sign_doc`.`with_bg` AS `proced_with_bg`,
    `con_sub`.`con_contact_name` AS `con_contact_name`,
    `con_sub`.`con_address` AS `con_address`,
    `inv`.`invoice_no` AS `invoice_no`,
    `inv`.`invoice_date` AS `invoice_date`,
    `inv`.`invoice_value` AS `invoice_value`,
    `inv`.`invoice_qty` AS `invoice_qty`,
    `inv`.`consignee_unique_id` AS `consignee_unique_id`,
    `inv`.`bill_status` AS `bill_status`,
    `po`.`file_name` AS `file_name`,
    `po`.`file_org_name` AS `file_org_name`,
    IFNULL(`po`.`ins_reqired`, 0) AS `ins_reqired`,
    IFNULL(`po`.`bank_required`, 0) AS `bank_required`
FROM
    (
        (
            (
                `aed_elcot`.`invoice_creation_main` `inv`
            JOIN `aed_elcot`.`po_form` `po`
            ON
                (
                    `inv`.`form_main_unique_id` = `po`.`unique_id`
                )
            )
        JOIN `aed_elcot`.`sign_doc_verification_detail` `sign_doc`
        ON
            (
                `sign_doc`.`form_main_unique_id` = `inv`.`form_main_unique_id` AND `sign_doc`.`con_unique_id` = `inv`.`consignee_unique_id`
            )
        )
    JOIN `aed_elcot`.`consignee_details_sub` `con_sub`
    ON
        (
            `con_sub`.`unique_id` = `inv`.`consignee_unique_id`
        )
    )
WHERE
    `sign_doc`.`status_app` = '2'
